# Transactional installation: keep old ZIP and manifest until both new files exist.
TX_ACTIVE=0
rollback_install(){
  (( ${TX_ACTIVE:-0} )) || return 0
  # Backup paths are authoritative if interruption occurs between mv and flag updates.
  if [[ -f "${TX_BZIP:-}" ]]; then
    rm -f -- "$TX_ZIP" || true
    mv -f -- "$TX_BZIP" "$TX_ZIP" || log 'CRITICAL: Could not restore original ZIP; backup retained'
  elif [[ "${TX_ZIP_MOVED:-0}" == 1 ]]; then
    rm -f -- "$TX_ZIP" || true
  fi
  if [[ -f "${TX_BMANIFEST:-}" ]]; then
    rm -f -- "$TX_MANIFEST" || true
    mv -f -- "$TX_BMANIFEST" "$TX_MANIFEST" || log 'CRITICAL: Could not restore original manifest; backup retained'
  elif [[ "${TX_MANIFEST_MOVED:-0}" == 1 ]]; then
    rm -f -- "$TX_MANIFEST" || true
  fi
  TX_ACTIVE=0
}
install_archive(){
  local source_zip=$1 source_manifest=$2 final_zip=$3 final_manifest=$4
  local backup_zip="${final_zip}.previous.${RUN}"
  local backup_manifest="${final_manifest}.previous.${RUN}"
  [[ ! -e "$backup_zip" && ! -e "$backup_manifest" ]] || return 1
  TX_ZIP=$final_zip TX_MANIFEST=$final_manifest TX_BZIP=$backup_zip TX_BMANIFEST=$backup_manifest
  TX_ZIP_MOVED=0 TX_MANIFEST_MOVED=0 TX_ACTIVE=1
  if [[ -e "$final_zip" ]]; then
    mv -- "$final_zip" "$backup_zip" || { rollback_install; return 1; }
  fi
  if [[ -e "$final_manifest" ]]; then
    mv -- "$final_manifest" "$backup_manifest" || { rollback_install; return 1; }
  fi
  if ! mv -- "$source_zip" "$final_zip"; then rollback_install; return 1; fi
  TX_ZIP_MOVED=1
  if ! mv -- "$source_manifest" "$final_manifest"; then rollback_install; return 1; fi
  TX_MANIFEST_MOVED=1
  TX_ACTIVE=0
  rm -f -- "$backup_zip" "$backup_manifest" || log 'WARNING: Old backup files could not be removed'
  return 0
}
# Define session cleanup before the EXIT trap can run on invalid credentials.
JSESSION=''
close_session(){
  if [[ -n "${JSESSION:-}" ]]; then
    curl -sS --connect-timeout 3 --max-time 8 -o /dev/null \
      -X DELETE -b "JSESSIONID=$JSESSION" "$HOST/data/JSESSION/" 2>/dev/null || true
    JSESSION=''
  fi
}

}

# Reuse one server session until it actually expires. Never refresh by a timer.
login(){
  local token status curl_rc=0
  # One authentication attempt per login. Keep body out of logs and stdout.
  AUTHRESP=$(mktemp "$ROOT/.cnda_login.XXXXXX") || { log 'AUTH ERROR: Cannot create secure login response file'; summary; exit 1; }
  status=$(curl -sS --connect-timeout 20 --max-time 90 -K "$CURLCFG" \
    -o "$AUTHRESP" -w '%{http_code}' "$HOST/REST/JSESSION") || curl_rc=$?
  if ((curl_rc != 0)); then
    case "$curl_rc" in
      6) log 'NETWORK ERROR: Could not resolve CNDA hostname (DNS)' ;;
      7) log 'NETWORK ERROR: Could not connect to CNDA' ;;
      28) log 'NETWORK ERROR: CNDA connection or response timed out' ;;
      35|51|60) log "TLS ERROR: Could not establish a trusted secure connection to CNDA (curl $curl_rc)" ;;
      *) log "NETWORK/TRANSFER ERROR: Login request failed (curl $curl_rc, HTTP ${status:-000})" ;;
    esac
    JSESSION=''; summary; exit 1
  fi
  case "$status" in
    200) ;;
    401) log 'AUTH ERROR: CNDA rejected the alias token or secret (HTTP 401). Check credentials or token expiration.' ; JSESSION=''; summary; exit 1 ;;
    403) log 'AUTH ERROR: CNDA refused the login (HTTP 403). Check account permissions or contact CNDA support.' ; JSESSION=''; summary; exit 1 ;;
    500|502|503|504) log "SERVER ERROR: CNDA login endpoint returned HTTP $status. Credentials are not confirmed invalid." ; JSESSION=''; summary; exit 1 ;;
    *) log "AUTH ERROR: Unexpected CNDA login response HTTP ${status:-000}. Credentials are not confirmed invalid." ; JSESSION=''; summary; exit 1 ;;
  esac
  token=$(cat "$AUTHRESP") || { log 'AUTH ERROR: Could not read login response'; summary; exit 1; }
  token=${token//$'\r'/}; token=${token//$'\n'/}
  [[ "$token" =~ ^[A-Za-z0-9._-]+$ ]] || {
    JSESSION=''; log 'SERVER RESPONSE ERROR: HTTP 200 but CNDA returned an invalid session token'; summary; exit 1;
  }
  JSESSION=$token
  rm -f "$AUTHRESP"
  AUTHRESP=''
}
# Request stores status in HTTP_CODE in the current shell so renewed sessions persist.
# On one HTTP 401, create one replacement session and retry that request once.
request(){
  local method=$1 url=$2 out=$3 attempt rc
  HTTP_CODE=000
  for attempt in 1 2; do
    if [[ "$method" == POST ]]; then
      HTTP_CODE=$(curl -sS -L --connect-timeout 20 --max-time 300 --retry 2 --retry-delay 3 \
        -b "JSESSIONID=$JSESSION" -H 'Content-Type: application/json' -X POST \
        --data-binary @"$BODY" -o "$out" -w '%{http_code}' "$url") || HTTP_CODE=000
    else
      HTTP_CODE=$(curl -sS -L --connect-timeout 20 --max-time 300 --retry 2 --retry-delay 3 \
        -b "JSESSIONID=$JSESSION" -o "$out" -w '%{http_code}' "$url") || HTTP_CODE=000
    fi
    if [[ "$HTTP_CODE" != 401 ]]; then return 0; fi
    if (( attempt == 1 )); then
      log 'Session expired (HTTP 401); obtaining one replacement session'
      JSESSION=''
      login
    fi
  done
  log 'AUTH ERROR: HTTP 401 persists after renewal'; summary; exit 1
}

# Discover scan types from the selected experiment, just as the CNDA download UI does.
# Do not silently request an archive with no selected scan types.
get_scan_types(){

import json,sys

rows=data.get('ResultSet',{}).get('Result',[])

types=sorted({str(row.get('type') or row.get('xnat_imagescandata_type') or '').strip()
              for row in rows if isinstance(row,dict)})
types=[t for t in types if t]
if not types:raise SystemExit('no scan types returned')
print(json.dumps(types))
PYSCANS
}
# Server returns a catalog ID as text or JSON; reject anything ambiguous.
catalog_id(){ python3 - "$1" <<'PY'
import json,re,sys
s=open(sys.argv[1]).read().strip()
try:
 o=json.loads(s)
 if isinstance(o,dict):s=str(o.get('id') or o.get('catalogId') or o.get('downloadId') or '')
 elif isinstance(o,str):s=o
except ValueError: pass
if not re.fullmatch(r'[A-Za-z0-9._-]+',s):raise SystemExit(1)
print(s)
PY
}
# Read optional estimated archive bytes from CNDA's size-aware response.
# Do not interpret an unknown response format as a valid estimate.
estimated_bytes(){ python3 - "$1" <<'PYEST'
import json,sys
try:
 with open(sys.argv[1],encoding='utf-8') as f: obj=json.load(f)
except (ValueError,OSError): raise SystemExit(1)
keys=('size','estimatedSize','estimated_size','downloadSize','download_size','totalSize','total_size','bytes','sizeInBytes')
def visit(o,depth=0):
 if depth>3 or not isinstance(o,dict):return None
 for key in keys:
  if key in o:
   v=o[key]
   if isinstance(v,int) and not isinstance(v,bool) and v>0:return v
   if isinstance(v,str) and v.isdecimal() and int(v)>0:return int(v)
 for key in ('data','result','download','catalog'):
  if key in o:
   v=visit(o[key],depth+1)
   if v:return v
 return None
n=visit(obj)
if not n:raise SystemExit(1)
print(n)
PYEST
}
# Print a single readable status line, using estimates only when provided.
transfer_status(){ python3 - "$1" "$2" "$3" <<'PYSTAT'
import sys
received=float(sys.argv[1]);elapsed=max(float(sys.argv[2]),0.001)
try: estimate=int(sys.argv[3])
except ValueError:estimate=0
def duration(sec):
 sec=max(0,int(sec)); h,r=divmod(sec,3600);m,s=divmod(r,60)
 return f'{h}h {m:02d}m {s:02d}s' if h else (f'{m}m {s:02d}s' if m else f'{s}s')
def size(n):
 for unit in ('B','KiB','MiB','GiB','TiB'):
  if n<1024 or unit=='TiB':return f'{n:.1f} {unit}' if unit!='B' else f'{int(n)} B'
  n/=1024
rate=received/elapsed
parts=[duration(elapsed)]
if estimate>0:
 parts.extend([f'{size(received)} / {size(estimate)}',f'~{min(100,100*received/estimate):.0f}%'])
else:parts.append(f'{size(received)} received')
parts.append(f'{size(rate)}/s')
if estimate>0 and rate>0 and received<estimate:parts.append('ETA '+duration((estimate-received)/rate))
elif estimate>0 and received>=estimate:parts.append('ETA calculating')
print(' | '.join(parts),file=sys.stderr,flush=True)
PYSTAT
}
